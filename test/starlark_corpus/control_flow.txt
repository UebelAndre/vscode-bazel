================================================================================
If statements
================================================================================

if a:
    b
    c

if a:
    b
elif c:
    d
elif e:
    f
else:
    g

if a: b

================================================================================
Nested if statements
================================================================================

if a:
    if b:
        c
    else:
        if e:
            f
g

================================================================================
For statements
================================================================================

for x in xs:
    print(x)

for k, v in d.items():
    print(k, v)

for (a, b) in pairs:
    pass

for [a, b] in pairs:
    pass

for x in (a, b):
    pass

for (x,) in [(1,), (2,)]:
    x

================================================================================
List comprehensions
================================================================================

[x for x in xs]
[a + b for (a, b) in items]
[a for b in c for a in b]
[(x, y) for x in xs for y in ys if x if y]
[a if b else c for a in d]

================================================================================
Dictionary comprehensions
================================================================================

{a: b for a, b in items}
{k: v for k in ks for v in vs if k != v}

================================================================================
Conditional expressions
================================================================================

a = b if c else d
f() if a else d
a if b else c if d else e
